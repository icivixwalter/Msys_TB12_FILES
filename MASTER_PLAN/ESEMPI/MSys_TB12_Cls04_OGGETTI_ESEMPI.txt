Public Sub ESEMPIO_OGG_01_Elenco()
Dim og As New MSys_TB12_Cls04_OGGETTI, v As Variant, sDb_s As String
    sDb_s = "C:\CASA\LINGUAGGI\ACCESS\PROGETTI_MDB\MSYS\MSys_TB40\MDB\GE_MSys_TB40.mdb"
    For Each v In og.ElencoOggetti("TUTTI", "MSys_TB40_*", sDb_s)

'--------------------------------------------------------------------------------------------------
' ESEMPIO 2 - portare nel database corrente gli oggetti di un mini-progetto (form e query)
'--------------------------------------------------------------------------------------------------
Public Sub ESEMPIO_OGG_02_AccodaDaMiniProgetto()
Dim og As New MSys_TB12_Cls04_OGGETTI, sDb_s As String, n As Long
    sDb_s = "C:\CASA\LINGUAGGI\ACCESS\PROGETTI_MDB\MSYS\MSys_TB40\MDB\GE_MSys_TB40.mdb"
    n = og.AccodaOggetti(sDb_s, "QUERY", "MSys_TB40_*")          ' non sovrascrive quelli gia' presenti
    n = n + og.AccodaOggetti(sDb_s, "FORMS", "MSys_TB40_Frm*")
    MsgBox n & " oggetti importati. Dettagli nel log della classe.", vbInformation
End Sub

'--------------------------------------------------------------------------------------------------
' ESEMPIO 4 - inventario: scrivere gli oggetti nella tabella del mini-progetto TB03 (senza doppioni)
'--------------------------------------------------------------------------------------------------
Public Sub ESEMPIO_OGG_04_Inventario()
Dim og As New MSys_TB12_Cls04_OGGETTI, n As Long
    ' tabella predefinita: Msys_TB03_ELENCO_ObjProgetti (locale o collegata)
    ' campi predefiniti:   COD_PROGETTO_s, TIPOGGETTO_s, NOMEOGGETTO_s, DATAINS_d
    ' per un'altra tabella: og.TabellaElenco = "MiaTabella": og.CampoNome = "NOME" ...
    n = og.AccodaElencoInTabella("MSys", vbNullString, "TUTTI", "MSys_*")
    If n < 0 Then MsgBox og.UltimoErrore, vbExclamation Else Debug.Print n & " righe aggiunte"
End Sub
